*Google mengharuskan jembatan Apps Script agar website bisa menulis baris baru secara otomatis ke Spreadsheet Anja.

Cara Mengaktifkan Auto-Write (1 Menit)

  1. Buka Google Spreadsheet Skripsi Anja, lalu klik menu Ekstensi (Extensions) > Apps Script.
  2. Hapus kode bawaan, lalu Paste kode script di bawah ini dan klik ikon Simpan (Save).
  3. Klik Terapkan (Deploy) > Deployment baru, pilih jenis Aplikasi web (Web app), ubah Yang memiliki akses (Who has access) menjadi Siapa saja (Anyone), lalu salin URL Web App-nya ke kolom di atas.
function doPost(e) {
  var ss = SpreadsheetApp.openById("1kqJ7kWMP4z8S7RqfqADQFMcSGEnRKLS3YqcGb84vZLE");
  var sheet = ss.getSheets()[0];
  var data = (e && e.parameter && e.parameter.tgl_lahir)
    ? e.parameter
    : JSON.parse(e.postData.contents);

  if (sheet.getLastRow() === 0) {
    var headers = [
      "Timestamp", "Tanggal Lahir Balita", "Usia Ibu", "Pendidikan", "Pekerjaan", "Usia Balita",
      "P1","P2","P3","P4","P5","P6","P7","P8","P9","P10","Total Pengetahuan",
      "S1","S2","S3","S4","S5","S6","S7","S8","S9","S10","Total Sikap",
      "T1","T2","T3","T4","T5","Total Tindakan","P+S+T","Total Hasil","Persentase %"
    ];
    sheet.appendRow(headers);
  }

  var totalP = Number(data.total_P || 0);
  var totalS = Number(data.total_S || 0);
  var totalT = Number(data.total_T || 0);
  var totalHasil = totalP + totalS + totalT;
  var pst = totalP + " + " + totalS + " + " + totalT;
  var persen = Math.round((totalHasil / 55) * 100) + "%";

  var row = [
    data.timestamp, data.tgl_lahir, data.usia_ibu, data.pendidikan, data.pekerjaan, data.usia_balita,
    data.P1, data.P2, data.P3, data.P4, data.P5, data.P6, data.P7, data.P8, data.P9, data.P10, totalP,
    data.S1, data.S2, data.S3, data.S4, data.S5, data.S6, data.S7, data.S8, data.S9, data.S10, totalS,
    data.T1, data.T2, data.T3, data.T4, data.T5, totalT, pst, totalHasil, persen
  ];
  sheet.appendRow(row);
  sheet.getRange(sheet.getLastRow(), 35).setNumberFormat("@").setValue(pst);

  return ContentService.createTextOutput(JSON.stringify({ status: "success" }))
    .setMimeType(ContentService.MimeType.JSON);
}

function doGet(e) {
  var ss = SpreadsheetApp.openById("1kqJ7kWMP4z8S7RqfqADQFMcSGEnRKLS3YqcGb84vZLE");
  var sheet = ss.getSheets()[0];
  var values = sheet.getDataRange().getValues();
  var records = [];
  for (var i = 1; i < values.length; i++) {
    var r = values[i];
    if (r.join("").trim() === "") continue;
    records.push({
      timestamp: String(r[0] || ""), tgl_lahir: String(r[1] || ""),
      usia_ibu: String(r[2] || ""), pendidikan: String(r[3] || ""),
      pekerjaan: String(r[4] || ""), usia_balita: String(r[5] || ""),
      P1: r[6], P2: r[7], P3: r[8], P4: r[9], P5: r[10], P6: r[11], P7: r[12], P8: r[13], P9: r[14], P10: r[15], total_P: r[16],
      S1: r[17], S2: r[18], S3: r[19], S4: r[20], S5: r[21], S6: r[22], S7: r[23], S8: r[24], S9: r[25], S10: r[26], total_S: r[27],
      T1: r[28], T2: r[29], T3: r[30], T4: r[31], T5: r[32], total_T: r[33],
      p_plus_s_plus_t: String(r[34] || ""), total_hasil: r[35], persentase: String(r[36] || "")
    });
  }
  return ContentService.createTextOutput(JSON.stringify({ status: "success", records: records }))
    .setMimeType(ContentService.MimeType.JSON);
}
