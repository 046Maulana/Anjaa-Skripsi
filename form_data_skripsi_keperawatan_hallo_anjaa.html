Koneksi Otomatis Google Spreadsheets

Hubungkan form ini ke Google Sheets Anja agar setiap klik Simpan & Kirim langsung menambah baris baru otomatis.

*Jika URL belum diisi, setiap pengiriman data tetap otomatis tersimpan di Rekap Sesi dan tersalin ke Clipboard agar tinggal Paste (Ctrl+V) di Google Sheets.

Cara Mengaktifkan Auto-Write (1 Menit)

  1. Buka Google Spreadsheet Skripsi Anja, lalu klik menu Ekstensi (Extensions) > Apps Script.
  2. Hapus kode bawaan, lalu Paste kode script di bawah ini dan klik ikon Simpan (Save).
  3. Klik Terapkan (Deploy) > Deployment baru, pilih jenis Aplikasi web (Web app), ubah akses menjadi Siapa saja (Anyone), lalu salin URL Web App-nya ke kolom di atas.
function doPost(e) {
  var ss = SpreadsheetApp.openById("1kqJ7kWMP4z8S7RqfqADQFMcSGEnRKLS3YqcGb84vZLE");
  var sheet = ss.getSheets()[0];
  var data = JSON.parse(e.postData.contents);

  if (sheet.getLastRow() === 0) {
    var headers = [
      "Timestamp", "Tanggal Lahir Balita", "Usia Ibu", "Pendidikan", "Pekerjaan", "Usia Balita",
      "P1","P2","P3","P4","P5","P6","P7","P8","P9","P10","Total Pengetahuan",
      "S1","S2","S3","S4","S5","S6","S7","S8","S9","S10","Total Sikap",
      "T1","T2","T3","T4","T5","Total Tindakan"
    ];
    sheet.appendRow(headers);
  }

  var row = [
    data.timestamp, data.tgl_lahir, data.usia_ibu, data.pendidikan, data.pekerjaan, data.usia_balita,
    data.P1, data.P2, data.P3, data.P4, data.P5, data.P6, data.P7, data.P8, data.P9, data.P10, data.total_P,
    data.S1, data.S2, data.S3, data.S4, data.S5, data.S6, data.S7, data.S8, data.S9, data.S10, data.total_S,
    data.T1, data.T2, data.T3, data.T4, data.T5, data.total_T
  ];
  sheet.appendRow(row);

  return ContentService.createTextOutput(JSON.stringify({ status: "success" }))
    .setMimeType(ContentService.MimeType.JSON);
}
